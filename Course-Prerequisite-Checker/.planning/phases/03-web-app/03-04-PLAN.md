/**
 * CoursePicker — searchable combobox of SJSU CS catalog courses.
 *
 * Props:
 *   value: string|null            — currently selected course_id
 *   onSelect: (course_id) => void — called when user picks a course
 *   disabled: boolean             — disables while loading verdict
 */
import React, { useEffect, useRef, useState } from 'react'
import { fetchCatalog } from '../api.js'

export function CoursePicker({ value, onSelect, disabled = false }) {
  const [catalog, setCatalog] = useState([])        // [{course_id, name}]
  const [catalogError, setCatalogError] = useState(null)
  const [query, setQuery] = useState('')
  const [open, setOpen] = useState(false)
  const containerRef = useRef(null)

  // Load catalog once on mount
  useEffect(() => {
    fetchCatalog()
      .then(setCatalog)
      .catch(err => setCatalogError(err.message))
  }, [])

  // Close dropdown on outside click
  useEffect(() => {
    function onClickOutside(e) {
      if (containerRef.current && !containerRef.current.contains(e.target)) {
        setOpen(false)
      }
    }
    document.addEventListener('mousedown', onClickOutside)
    return () => document.removeEventListener('mousedown', onClickOutside)
  }, [])

  const filtered = query.trim()
    ? catalog.filter(c =>
        c.course_id.toLowerCase().includes(query.toLowerCase()) ||
        c.name.toLowerCase().includes(query.toLowerCase())
      )
    : catalog

  const selectedCourse = catalog.find(c => c.course_id === value)

  function handleSelect(courseId) {
    onSelect(courseId)
    setQuery('')
    setOpen(false)
  }

  function handleInputChange(e) {
    setQuery(e.target.value)
    setOpen(true)
  }

  function handleInputFocus() {
    setOpen(true)
  }

  if (catalogError) {
    return (
      <p className="text-sm text-red-600">
        Failed to load catalog: {catalogError}
      </p>
    )
  }

  return (
    <div ref={containerRef} className="relative">
      <div className={`
        flex items-center border rounded-lg px-3 py-2 gap-2 bg-white
        ${disabled ? 'opacity-50 pointer-events-none' : ''}
        ${open ? 'border-indigo-500 ring-2 ring-indigo-200' : 'border-gray-300'}
      `}>
        {/* Search icon */}
        <svg className="w-4 h-4 text-gray-400 flex-none" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2}
            d="M21 21l-4.35-4.35M17 11A6 6 0 105 11a6 6 0 0012 0z" />
        </svg>
        <input
          type="text"
          className="flex-1 text-sm outline-none bg-transparent placeholder-gray-400"
          placeholder={selectedCourse ? `${selectedCourse.course_id} — ${selectedCourse.name}` : 'Search CS courses…'}
          value={query}
          onChange={handleInputChange}
          onFocus={handleInputFocus}
          disabled={disabled}
        />
        {value && (
          <button
            onClick={() => { onSelect(null); setQuery('') }}
            className="text-gray-400 hover:text-gray-600"
            aria-label="Clear selection"
          >
            <svg className="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </button>
        )}
        <svg className="w-4 h-4 text-gray-400 flex-none" fill="none" stroke="currentColor" viewBox="0 0 24 24">
          <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2} d="M19 9l-7 7-7-7" />
        </svg>
      </div>

      {open && (
        <ul className="absolute z-50 mt-1 w-full bg-white border border-gray-200 rounded-lg shadow-lg max-h-64 overflow-y-auto">
          {filtered.length === 0 ? (
            <li className="px-4 py-3 text-sm text-gray-400">No courses match "{query}"</li>
          ) : (
            filtered.map(c => (
              <li
                key={c.course_id}
                onClick={() => handleSelect(c.course_id)}
                className={`
                  px-4 py-2.5 cursor-pointer flex items-center gap-3 text-sm
                  hover:bg-indigo-50 hover:text-indigo-700
                  ${value === c.course_id ? 'bg-indigo-50 text-indigo-700' : 'text-gray-700'}
                `}
              >
                <span className="font-mono font-semibold w-20 flex-none">{c.course_id}</span>
                <span className="truncate">{c.name}</span>
              </li>
            ))
          )}
        </ul>
      )}
    </div>
  )
}

export function VerdictCard({ courseId, courseName, verdict, loading, error }) {
  if (!courseId) {
    return (
      <div className="rounded-lg border border-dashed border-gray-200 p-8 text-center text-sm text-gray-400">
        Select a course above to see your eligibility verdict.
      </div>
    )
  }

  if (loading) {
    return (
      <div className="rounded-lg border border-gray-200 bg-white p-8 flex items-center justify-center gap-3 shadow-sm">
        <Spinner />
        <span className="text-sm text-gray-500">Checking eligibility…</span>
      </div>
    )
  }

  if (error) {
    return (
      <div className="rounded-lg border border-red-200 bg-red-50 p-6 shadow-sm">
        <p className="text-sm text-red-700 font-medium">Error checking eligibility</p>
        <p className="text-sm text-red-600 mt-1">{error}</p>
      </div>
    )
  }

  if (!verdict) return null

  const { eligible, explanation } = verdict

  // Highlight course IDs in explanation (e.g., CS46A, MATH42) with mono style
  function formatExplanation(text) {
    // Split on course-id-like tokens: 2-4 uppercase letters + digits + optional letters
    const parts = text.split(/\b([A-Z]{2,4}\d+[A-Z]?)\b/)
    return parts.map((part, i) =>
      /^[A-Z]{2,4}\d+[A-Z]?$/.test(part)
        ? <code key={i} className="font-mono text-indigo-700 bg-indigo-50 px-1 rounded text-xs">{part}</code>
        : part
    )
  }

  return (
    <div className={`rounded-lg border p-6 shadow-sm ${eligible ? 'border-green-200 bg-green-50' : 'border-red-200 bg-red-50'}`}>
      {/* Header */}
      <div className="flex items-center gap-3 mb-3">
        {eligible ? (
          <div className="flex-none w-8 h-8 rounded-full bg-green-600 flex items-center justify-center">
            <svg className="w-5 h-5 text-white" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2.5} d="M5 13l4 4L19 7" />
            </svg>
          </div>
        ) : (
          <div className="flex-none w-8 h-8 rounded-full bg-red-600 flex items-center justify-center">
            <svg className="w-5 h-5 text-white" fill="none" stroke="currentColor" viewBox="0 0 24 24">
              <path strokeLinecap="round" strokeLinejoin="round" strokeWidth={2.5} d="M6 18L18 6M6 6l12 12" />
            </svg>
          </div>
        )}
        <div>
          <p className={`text-sm font-bold uppercase tracking-wide ${eligible ? 'text-green-700' : 'text-red-700'}`}>
            {eligible ? 'Eligible' : 'Not Eligible'}
          </p>
          <p className="text-base font-semibold text-gray-900">
            <code className="font-mono">{courseId}</code>
            {courseName && <span className="text-gray-500 font-normal ml-2">— {courseName}</span>}
          </p>
        </div>
      </div>

      {/* Explanation */}
      <p className="text-sm text-gray-700 leading-relaxed">
        {formatExplanation(explanation)}
      </p>
    </div>
  )
}

function AppContent() {
  const { backend } = useBackend()

  // Student record state
  const [completed, setCompleted] = useState({})

    return () => { cancelled = true }
  }, [selectedCourse, backend, completed, inProgress, hasTranscript])

  // Fire recommendations whenever backend or completed changes (after transcript uploaded)
  useEffect(() => {
    if (!hasTranscript) return

    let cancelled = false
    setRecsLoading(true)
    setRecsError(null)

    fetchRecommendations(completed, inProgress, backend)
      .then(result => { if (!cancelled) setRecommendations(result) })
      .catch(err => { if (!cancelled) setRecsError(err.message) })
      .finally(() => { if (!cancelled) setRecsLoading(false) })

    return () => { cancelled = true }
  }, [backend, completed, inProgress, hasTranscript])

  async function handleUpload(file) {
    setUploadLoading(true)
    setUploadError(null)
    try {
      const record = await uploadTranscript(file)
      setCompleted(record.completed ?? {})
      setInProgress(record.in_progress ?? [])
      setHasTranscript(true)
      // Reset downstream state
      setSelectedCourse(null)
      setVerdict(null)
      setRecommendations(null)
    } catch (err) {
      setUploadError(err.message ?? 'Upload failed. Please try again.')
    } finally {
      setUploadLoading(false)
    }
  }

  function handleCourseSelect(courseId) {
    setSelectedCourse(courseId)
    setVerdict(null)
  }

  return (
    <main className="max-w-5xl mx-auto px-4 py-8 space-y-8">
      {/* Step 1 — Upload */}
      <section>
        <SectionHeader step="1" title="Upload your transcript" />
        <UploadCard
          onUpload={handleUpload}
          loading={uploadLoading}
          error={uploadError}
        />
      </section>

      {hasTranscript && (
        <>
          {/* Step 2 — Confirm courses */}
          <section>
            <SectionHeader step="2" title="Confirm your courses" />
            <CourseList courses={completed} onChange={setCompleted} />
            {inProgress.length > 0 && (
              <p className="mt-3 text-sm text-gray-500">
                <span className="font-medium text-gray-700">In progress:</span>{' '}
                {inProgress.join(', ')}
              </p>
            )}
          </section>

          {/* Step 3 — Pick a course + verdict */}
          <section>
            <SectionHeader step="3" title="Pick a target course" />
            <CoursePicker
              value={selectedCourse}
              onSelect={handleCourseSelect}
              disabled={verdictLoading}
            />
            <div className="mt-4">
              <VerdictCard
                courseId={selectedCourse}
                courseName={selectedCourseName}
                verdict={verdict}
                loading={verdictLoading}
                error={verdictError}
              />
            </div>
          </section>

          {/* Step 4 — Recommendations */}
          <section>
            <SectionHeader step="4" title="Recommended next courses" />
            <RecommendationsPanel
              recommendations={recommendations}
              loading={recsLoading}
              error={recsError}
              onSelectCourse={courseId => {
                handleCourseSelect(courseId)
                // Scroll to Step 3
                document.getElementById('step3-section')?.scrollIntoView({ behavior: 'smooth' })
              }}
            />
          </section>
        </>
      )}
    </main>
  )
}

export default function App() {
  return (

      <div className="min-h-screen bg-gray-50 text-gray-900 font-sans">
        <TopNav />
        <AppContent />
      </div>
    </BackendProvider>
  )
}
